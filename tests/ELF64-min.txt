# Linux: Minimal ELF 64-bit - Hello World example
# =================== 1. ELF64 HEADER (64 байта) ===================
7F 45 4C 46                                      # e_ident: Магические байты ("\x7FELF")
02                                               # e_ident[EI_CLASS]: 64-битный формат (ELFCLASS64)
01                                               # e_ident[EI_DATA]: Little-endian (ELFDATA2LSB)
01                                               # e_ident[EI_VERSION]: Текущая версия ELF
00                                               # e_ident[EI_OSABI]: System V ABI
00                                               # e_ident[EI_ABIVERSION]: Версия ABI
00 00 00 00 00 00 00                             # Padding: Заполнители нулями (до 16 байт)

02 00                                            # e_type: Исполняемый файл (ET_EXEC)
3E 00                                            # e_machine: AMD x86-64
01 00 00 00                                      # e_version: Версия 1
78 00 40 00 00 00 00 00                          # e_entry: Точка входа (виртуальный адрес 0x400078)
40 00 00 00 00 00 00 00                          # e_phoff: Смещение таблицы программных заголовков (0x40)
00 00 00 00 00 00 00 00                          # e_shoff: Смещение таблицы секций (0 = нет секций)
00 00 00 00                                      # e_flags: Флаги процессора (0)
40 00                                            # e_ehsize: Размер этого заголовка (64 байта)
38 00                                            # e_phentsize: Размер одной записи Program Header (56 байт)
01 00                                            # e_phnum: Количество записей Program Header (1)
00 00                                            # e_shentsize: Размер Section Header (0)
00 00                                            # e_shnum: Количество Section Header (0)
00 00                                            # e_shstrndx: Индекс секции со строками (0)

# =================== 2. PROGRAM HEADER TABLE (56 байт) ============
01 00 00 00                                      # p_type: PT_LOAD (загружаемый сегмент)
05 00 00 00                                      # p_flags: PF_R (read) + PF_X (execute)
00 00 00 00 00 00 00 00                          # p_offset: Смещение в файле (0)
00 00 40 00 00 00 00 00                          # p_vaddr: Виртуальный адрес в памяти (0x400000)
00 00 40 00 00 00 00 00                          # p_paddr: Физический адрес (0x400000)
B2 00 00 00 00 00 00 00                          # p_filesz: Размер файла на диске (0xB2 байт)
B2 00 00 00 00 00 00 00                          # p_memsz: Размер сегмента в ОЗУ (0xB2 байт)
00 00 20 00 00 00 00 00                          # p_align: Выравнивание (2MB)

# =================== 3. ТЕЛО ПРОГРАММЫ (_start) — Смещение 0x78 в файле (VMA 0x400078)
# --- Системный вызов sys_write (вывод строки "Hello World!") ---
48 C7 C0 01 00 00 00                             # mov rax, 1 (системный вызов sys_write)
48 C7 C7 01 00 00 00                             # mov rdi, 1 (дескриптор stdout = 1)
48 BE A5 00 40 00 00 00 00 00                    # mov rsi, 0x4000A5 (указатель на строку msg)
48 C7 C2 0E 00 00 00                             # mov rdx, 14 (длина строки в байтах)
0F 05                                            # syscall (вызов ядра Linux)

# --- Системный вызов sys_exit (выход из программы) ---
48 C7 C0 3C 00 00 00                             # mov rax, 60 (системный вызов sys_exit, 60 в Linux x86_64)
48 31 FF                                         # xor rdi, rdi (код возврата 0)
0F 05                                            # syscall (вызов ядра Linux)

# --- Данные: Строка сообщения (VMA 0x4000A5) ---
48 65 6C 6C 6F 20 57 6F 72 6C 64 21 0A 00        # "Hello World!\n\0"
